# V0.5 - 高频数据持久化

> **目标**：实现高频位置数据的异步持久化，不阻塞游戏逻辑
> **存储方案**：Redis (高频缓存) + MySQL (基础数据)
> **完成时间**：2026-05-06

---

## 数据分层

| 层级 | 存储 | 用途 |
|------|------|------|
| 基础数据 | MySQL | 账号、角色、属性 |
| 高频缓存 | SceneServer 内存 | 当前位置 |
| 位置快照 | Redis | 断线恢复、跨服同步 |

---

## Redis 架构

```
SceneServer ──异步──▶ KVServant (DBServer) ──▶ Redis
                │
                └──同步──▶ 断线保存、重连读取
```

### Redis 键值

```
player:kv:{playerId}     # HSET 存储位置快照
```

### Redis 容器

```bash
docker run -d --name tars-redis \
  --network tars \
  -p 6379:6379 \
  redis:latest redis-server --appendonly yes
```

## 接口设计

### KVServant (DBServer)

```tars
// ========== 高频数据错误码 ==========
const int ERR_KV_NOT_FOUND = 6001;

// ========== 高频玩家数据 ==========
struct PlayerKVData
{
    0 require long playerId;       // 玩家ID
    1 require int level;            // 等级
    2 require float x;              // 位置X
    3 require float y;              // 位置Y
    4 require float z;              // 位置Z
    5 require int sceneId;          // 场景ID
    6 optional long updateTime;     // 更新时间
};

interface KVServant
{
    int savePlayerData(PlayerKVData data);          // 同步保存
    int batchSavePlayerData(vector<PlayerKVData> dataList);  // 批量保存
    int getPlayerData(long playerId, out PlayerKVData data); // 获取
    int deletePlayerData(long playerId);            // 删除
    int batchGetPlayerData(vector<long> playerIds, out vector<PlayerKVData> dataList); // 批量获取
};
```

---

## 调用策略

| 场景 | 调用方式 | 说明 |
|------|----------|------|
| enterScene | getPlayerData + savePlayerData | 恢复/初始化 |
| leaveScene | **savePlayerData (同步)** | 必须同步 |
| 断线 | **savePlayerData (同步)** | 防御性落盘 |
| 位置移动 | async_savePlayerData | 高频异步 |

### Tars 异步调用

```cpp
// 同步 - 阻塞等待，适合断线保存、重连读取
kvPrx->savePlayerData(data);

// 异步 - 发完即忘，不阻塞游戏逻辑
kvPrx->async_savePlayerData(NULL, data);
```

## 恢复策略

```
restoreOnEnter(playerId, targetSceneId)

┌─────────────────────────────────────────────┐
│  1. 检查内存中是否存在                        │
│     └─ 存在且同场景 → setOnline 在线          │
│                                             │
│  2. 从 Redis 获取数据                        │
│     └─ 不存在 → 新玩家初始化                  │
│                                             │
│  3. 判断场景关系                             │
│     ├─ 同场景 → 恢复位置 (x,y,z)             │
│     └─ 跨场景 → 入口点生成                   │
└─────────────────────────────────────────────┘
```

| 场景 | 位置来源 | 操作 |
|------|----------|------|
| 新玩家 | 入口点 (0,0,0) | 初始化 + 同步保存 |
| 同场景离线恢复 | Redis | 恢复原位置 |
| 跨场景 | 入口点 (0,0,0) | 重新生成 + 同步保存 |

## Redis 实现 (KVImp.cpp)

```cpp
// ========== KVImp.h ==========
class KVImp : public KVServant
{
private:
    redisContext* _redis;
    std::string _redisHost;
    int _redisPort;
    int _dataTTL;  // 秒

public:
    int savePlayerData(const PlayerKVData& data, tars::TarsCurrentPtr current) override;
    int batchSavePlayerData(const vector<PlayerKVData>& dataList, tars::TarsCurrentPtr current) override;
    int getPlayerData(tars::Int64 playerId, PlayerKVData& data, tars::TarsCurrentPtr current) override;
    int deletePlayerData(tars::Int64 playerId, tars::TarsCurrentPtr current) override;
    int batchGetPlayerData(const vector<tars::Int64>& playerIds, vector<PlayerKVData>& dataList, tars::TarsCurrentPtr current) override;

private:
    std::string playerKey(tars::Int64 playerId);
};
```

```cpp
// ========== savePlayerData ==========
int KVImp::savePlayerData(const PlayerKVData& data, tars::TarsCurrentPtr current)
{
    if (!_redis) return -1;
    
    string key = playerKey(data.playerId);
    redisReply *reply = (redisReply*)redisCommand(_redis,
        "HSET %s level %d x %f y %f z %f sceneId %d updateTime %ld",
        key.c_str(), data.level, data.x, data.y, data.z, data.sceneId, data.updateTime);
    
    if (reply) freeReplyObject(reply);
    redisCommand(_redis, "EXPIRE %s %d", key.c_str(), _dataTTL);
    return 0;
}

// ========== getPlayerData ==========
int KVImp::getPlayerData(tars::Int64 playerId, PlayerKVData& data, tars::TarsCurrentPtr current)
{
    if (!_redis) return -1;
    
    redisReply *reply = (redisReply*)redisCommand(_redis, "HGETALL %s", playerKey(playerId).c_str());
    if (!reply || reply->type != REDIS_REPLY_ARRAY || reply->elements == 0) {
        if (reply) freeReplyObject(reply);
        return ERR_KV_NOT_FOUND;
    }
    
    data.playerId = playerId;
    for (size_t i = 0; i < reply->elements; i += 2) {
        string field = reply->element[i]->str;
        string value = reply->element[i + 1]->str;
        
        if (field == "level") data.level = TC_Common::strto<int>(value);
        else if (field == "x") data.x = TC_Common::strto<float>(value);
        else if (field == "y") data.y = TC_Common::strto<float>(value);
        else if (field == "z") data.z = TC_Common::strto<float>(value);
        else if (field == "sceneId") data.sceneId = TC_Common::strto<int>(value);
        else if (field == "updateTime") data.updateTime = TC_Common::strto<long>(value);
    }
    
    freeReplyObject(reply);
    return 0;
}
```

## SceneServer 集成

```cpp
// PlayerManager.h
class PlayerManager
{
private:
    KVServantPrx _kvPrx;

public:
    void setKVPrx(const KVServantPrx& kvPrx);
    bool restoreOnEnter(int64_t playerId, int32_t targetSceneId);  // 重连恢复
    void onPositionChange(int64_t playerId, float x, float y, float z);  // 异步保存
    void onPlayerLeave(int64_t playerId);  // 同步保存后清除
    void onPlayerOffline(int64_t playerId);  // 同步保存
};
```

```cpp
// ========== restoreOnEnter ==========
bool PlayerManager::restoreOnEnter(int64_t playerId, int32_t targetSceneId)
{
    if (!_kvPrx) return false;

    PlayerKVData dbData;
    int ret = _kvPrx->getPlayerData(playerId, dbData);
    
    if (ret != 0) {
        // 新玩家：初始化
        playerEnter(playerId, 0.0f, 0.0f, 0.0f, targetSceneId, 1);
        savePlayerDataSync(playerId);
        return false;
    }

    // 检查内存
    bool inMemory = (getPlayer(playerId) != nullptr);
    int32_t currentSceneId = inMemory ? getPlayer(playerId)->sceneId : dbData.sceneId;

    if (inMemory && currentSceneId == targetSceneId) {
        setOnline(playerId, true);
        return false;
    }

    // 同场景恢复
    if (currentSceneId == targetSceneId) {
        playerEnter(playerId, dbData.x, dbData.y, dbData.z, dbData.sceneId, dbData.level);
        return false;
    }

    // 跨场景：入口点生成
    if (inMemory) removeFromAOI(playerId);
    playerEnter(playerId, 0.0f, 0.0f, 0.0f, targetSceneId, dbData.level);
    savePlayerDataSync(playerId);
    return true;
}

// ========== 异步保存位置 ==========
void PlayerManager::onPositionChange(int64_t playerId, float x, float y, float z)
{
    auto* player = getPlayer(playerId);
    if (!player) return;
    
    player->x = x; player->y = y; player->z = z;
    
    PlayerKVData data;
    data.playerId = playerId;
    data.x = x; data.y = y; data.z = z;
    data.sceneId = player->sceneId;
    data.level = player->level;
    data.updateTime = time(NULL);
    
    _kvPrx->async_savePlayerData(NULL, data);
}

// ========== 同步保存 ==========
void PlayerManager::savePlayerDataSync(int64_t playerId)
{
    if (!_kvPrx) return;
    
    auto* player = getPlayer(playerId);
    if (!player) return;

    PlayerKVData data;
    data.playerId = playerId;
    data.x = player->x; data.y = player->y; data.z = player->z;
    data.sceneId = player->sceneId;
    data.level = player->level;
    data.updateTime = time(NULL);

    int ret = _kvPrx->savePlayerData(data);
    if (ret == 0) {
        TLOG_DEBUG("savePlayerDataSync playerId=" << playerId << " saved" << endl);
    } else {
        TLOG_ERROR("savePlayerDataSync playerId=" << playerId << " failed" << endl);
    }
}
```
