## 部署文档

### 一键部署脚本
```bash
#!/bin/bash
# deploy.sh

# 1. 初始化数据库
mysql -h $MYSQL_HOST -u root -p < scripts/init_db.sql

# 2. 初始化 TcaplusDB
tcaplus-cli import -f scripts/player_pos.tcaplus

# 3. 部署服务
for server in LobbyServer SceneServer DBServer; do
    cd /workspace/server/$server/build
    make $server-upload
    # 发布到 Tars 平台
done

# 4. 验证服务
./scripts/health_check.sh
```
